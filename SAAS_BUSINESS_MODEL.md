// src/lib/tenant.ts
export class TenantService {
  // Isolamento de dados por empresa
  async getTenantData(tenantId: string, userId: string) {
    return prisma.user.findMany({
      where: {
        companyId: tenantId,
        // Row-level security
      }
    })
  }
  
  // Limites por plano
  async checkPlanLimits(tenantId: string, resource: string) {
    const company = await prisma.company.findUnique({
      where: { id: tenantId },
      include: { subscription: true }
    })
    
    const plan = company.subscription.plan
    const limits = PLAN_LIMITS[plan]
    
    return this.validateLimit(resource, limits)
  }
}
```

// src/services/billing/usage.service.ts
export class UsageService {
  async trackUsage(tenantId: string, resource: string, quantity: number) {
    await prisma.usage.create({
      data: {
        tenantId,
        resource,
        quantity,
        month: new Date().getMonth(),
        year: new Date().getFullYear()
      }
    })
    
    // Check overage
    await this.checkOverage(tenantId, resource)
  }
  
  async generateInvoice(tenantId: string, month: number, year: number) {
    const usage = await this.getMonthlyUsage(tenantId, month, year)
    const subscription = await this.getSubscription(tenantId)
    
    return this.calculateBilling(usage, subscription)
  }
}
```

### 🔐 **Plan-based Access Control:**

```typescript
// src/middleware/plan-access.ts
export function planAccessMiddleware(requiredPlan: PlanType) {
  return async (req: NextRequest) => {
    const session = await getServerSession(authConfig)
    const company = await prisma.company.findUnique({
      where: { id: session.user.companyId },
      include: { subscription: true }
    })
    
    const userPlan = company.subscription.plan
    const hasAccess = PLAN_HIERARCHY[userPlan] >= PLAN_HIERARCHY[requiredPlan]
    
    if (!hasAccess) {
      return NextResponse.json({ error: 'Upgrade required' }, { status: 402 })
    }
    
    return NextResponse.next()
  }
}
```
